<script>
// =====================================================
// 1. QUY ƯỚC 6 MẶT
// =====================================================
const faces = [
  { code: 'U', name: 'Up', color: 'W', index: 0 },
  { code: 'L', name: 'Left', color: 'R', index: 1 },
  { code: 'F', name: 'Front', color: 'B', index: 2 },
  { code: 'R', name: 'Right', color: 'O', index: 3 },
  { code: 'B', name: 'Back', color: 'G', index: 4 },
  { code: 'D', name: 'Down', color: 'Y', index: 5 }
];

// =====================================================
// 2. MÀU HIỂN THỊ
// =====================================================
const colorClasses = {
  'W': 'c-white',
  'R': 'c-red',
  'B': 'c-blue',
  'O': 'c-orange',
  'G': 'c-green',
  'Y': 'c-yellow'
};

// =====================================================
// 3. MÀU → KÝ TỰ KOCIEMBA
// =====================================================
const colorToKociemba = {
  'W': 'U',
  'O': 'R',
  'B': 'F',
  'Y': 'D',
  'R': 'L',
  'G': 'B'
};

// =====================================================
// 4. BIẾN
// =====================================================
let selectedColor = 'W';
let cubeData = new Array(54).fill('');

// =====================================================
// 5. KHỞI TẠO
// =====================================================
function init() {
  createAllFaces();
  setCenterColors(); // Tô màu sẵn cho các tâm mặt
  selectPaletteColor('W');
  renderAllFaces();
  updatePaletteCounts();
  checkCompletion();
}

// Gán cố định màu tâm cho 6 mặt
function setCenterColors() {
  faces.forEach(face => {
    const centerGlobalIndex = face.index * 9 + 4; // Ô chính giữa (Index 4)
    cubeData[centerGlobalIndex] = face.color;
  });
}

// Nút Reset xoá màu toàn bộ trừ các tâm
function resetCube() {
  cubeData = new Array(54).fill('');
  setCenterColors();
  renderAllFaces();
  updatePaletteCounts();
  checkCompletion();
}

// =====================================================
// 6. TẠO 6 LƯỚI 3×3
// =====================================================
function createAllFaces() {
  faces.forEach(face => {
    const container = document.getElementById('face-' + face.code);
    container.innerHTML = '';

    for (let i = 0; i < 9; i++) {
      const cell = document.createElement('div');
      cell.className = 'grid-cell striped-bg';

      // Khoá không cho tô đè lên tâm nếu cần, hoặc tô bình thường
      cell.onclick = function() {
        paintCell(face.index, i);
      };

      container.appendChild(cell);
    }
  });
}

// =====================================================
// 7. CHỌN MÀU
// =====================================================
function selectPaletteColor(colorCode) {
  selectedColor = colorCode;
  document.querySelectorAll('.color-circle').forEach(el => el.classList.remove('selected'));
  document.getElementById('btn-' + colorCode).classList.add('selected');
}

// =====================================================
// 8. TÔ Ô
// =====================================================
function paintCell(faceIndex, cellIndex) {
  const globalIndex = faceIndex * 9 + cellIndex;
  const oldColor = cubeData[globalIndex];

  if (oldColor !== selectedColor) {
    const countNewColor = cubeData.filter(c => c === selectedColor).length;
    if (countNewColor >= 9) {
      alert('Màu này đã đủ 9 ô!');
      return;
    }
  }

  cubeData[globalIndex] = selectedColor;
  renderAllFaces();
  updatePaletteCounts();
  checkCompletion();
}

// =====================================================
// 9. HIỂN THỊ 6 MẶT
// =====================================================
function renderAllFaces() {
  faces.forEach(face => {
    const cells = document.getElementById('face-' + face.code).querySelectorAll('.grid-cell');

    for (let i = 0; i < 9; i++) {
      const globalIndex = face.index * 9 + i;
      const colorCode = cubeData[globalIndex];

      // Mặc định trả về sọc chéo
      cells[i].className = 'grid-cell striped-bg';

      if (colorCode && colorClasses[colorCode]) {
        cells[i].classList.add(colorClasses[colorCode]);
      }
    }
  });
}

// =====================================================
// 10. ĐẾM MÀU
// =====================================================
function updatePaletteCounts() {
  ['W', 'R', 'B', 'O', 'G', 'Y'].forEach(color => {
    const used = cubeData.filter(c => c === color).length;
    const remain = 9 - used;
    document.getElementById('count-' + color).innerText = remain;
  });
}

// =====================================================
// 11. KIỂM TRA ĐỦ 54 Ô
// =====================================================
function checkCompletion() {
  const filledCount = cubeData.filter(c => c !== '').length;
  const btnExport = document.getElementById('btnExport');
  const resultBox = document.getElementById('resultBox');

  if (filledCount === 54) {
    btnExport.style.display = 'block';
    resultBox.innerHTML = `
      <b>✅ Đã nhập đủ 54 ô.</b> Sẵn sàng tính toán!
    `;
  } else {
    btnExport.style.display = 'none';
    resultBox.innerHTML = `
      <b>Trạng thái:</b> Đã nhập ${filledCount} / 54 ô.
    `;
  }
}

// =====================================================
// 12. LẤY 1 MẶT 3×3
// =====================================================
function getFace(faceIndex) {
  let face = [];
  for (let i = 0; i < 9; i++) {
    face.push(cubeData[faceIndex * 9 + i]);
  }
  return face;
}

// =====================================================
// 13. CHUYỂN 1 MẶT → KOCIEMBA
// =====================================================
function convertFaceToKociemba(face) {
  let result = '';
  for (let i = 0; i < 9; i++) {
    const color = face[i];
    const letter = colorToKociemba[color];
    if (!letter) return null;
    result += letter;
  }
  return result;
}

// =====================================================
// 14. CHUYỂN TOÀN BỘ → URFDLB
// =====================================================
function convertToKociemba() {
  const Up    = getFace(0);
  const Left  = getFace(1);
  const Front = getFace(2);
  const Right = getFace(3);
  const Back  = getFace(4);
  const Down  = getFace(5);

  const U = convertFaceToKociemba(Up);
  const R = convertFaceToKociemba(Right);
  const F = convertFaceToKociemba(Front);
  const D = convertFaceToKociemba(Down);
  const L = convertFaceToKociemba(Left);
  const B = convertFaceToKociemba(Back);

  if (!U || !R || !F || !D || !L || !B) {
    alert('LỖI: Tất cả 54 ô phải được nhập màu!');
    return '';
  }

  const kociembaStr = U + R + F + D + L + B;

  if (kociembaStr.length !== 54) {
    alert('LỖI: Chuỗi Kociemba phải có 54 ký tự!');
    return '';
  }

  for (let i = 0; i < kociembaStr.length; i++) {
    const ch = kociembaStr[i];
    if (ch !== 'U' && ch !== 'R' && ch !== 'F' && ch !== 'D' && ch !== 'L' && ch !== 'B') {
      alert('LỖI: Chuỗi Kociemba có ký tự không hợp lệ!');
      return '';
    }
  }

  const count = { U: 0, R: 0, F: 0, D: 0, L: 0, B: 0 };
  for (let i = 0; i < kociembaStr.length; i++) {
    count[kociembaStr[i]]++;
  }

  for (const key of ['U', 'R', 'F', 'D', 'L', 'B']) {
    if (count[key] !== 9) {
      alert('LỖI: Ký tự ' + key + ' xuất hiện ' + count[key] + ' lần, phải là 9 lần!');
      return '';
    }
  }

  return kociembaStr;
}

// =====================================================
// 15. HIỂN THỊ SUMMARY (KHÔNG DÙNG TRỰC TIẾP TỰ ĐỘNG)
// =====================================================
function showFaceSummary() {
  const Up    = convertFaceToKociemba(getFace(0));
  const Left  = convertFaceToKociemba(getFace(1));
  const Front = convertFaceToKociemba(getFace(2));
  const Right = convertFaceToKociemba(getFace(3));
  const Back  = convertFaceToKociemba(getFace(4));
  const Down  = convertFaceToKociemba(getFace(5));

  if (!Up || !Left || !Front || !Right || !Back || !Down) return;

  const kociembaStr = Up + Right + Front + Down + Left + Back;

  document.getElementById('resultBox').innerHTML = `
    <b>✅ Chuỗi Kociemba:</b>
    <div class="kociemba-string">${kociembaStr}</div>
  `;
}

// =====================================================
// 16. XUẤT KOCIEMBA + GỬI ESP32
// =====================================================
function exportKociemba() {
  const kociembaStr = convertToKociemba();
  if (!kociembaStr) return;

  document.getElementById('resultBox').innerHTML = `
    <strong>Ngõ vào Kociemba:</strong>
    <div class="kociemba-string">${kociembaStr}</div>
    <div style="margin-top:4px;">
      <span id="solverStatus">⏳ Đang gửi lên Server...</span>
      <div id="kociembaResult" class="solution">-</div>
    </div>
  `;

  fetch('/solve_online?state=' + encodeURIComponent(kociembaStr))
  .then(response => response.json())
  .then(data => {
    if (data.status === "success") {
      document.getElementById('solverStatus').innerText = `✅ Thành công! (${data.time} ms)`;
      document.getElementById('kociembaResult').innerText = data.solution;
    } else {
      document.getElementById('solverStatus').innerText = '❌ Lỗi tính toán!';
      document.getElementById('kociembaResult').innerText = data.message || 'Không tìm thấy lời giải!';
    }
  })
  .catch(err => {
    document.getElementById('solverStatus').innerText = '⚠️ Không nhận được phản hồi Server.';
    document.getElementById('kociembaResult').innerText = 'Đã tạo chuỗi: ' + kociembaStr;

    fetch('/kociemba_output?str=' + encodeURIComponent(kociembaStr));
  });
}

// =====================================================
// 17. CHẠY KHI MỞ TRANG
// =====================================================
window.onload = init;
